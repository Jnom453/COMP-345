27void Game::buildMap()
28{
29 map = Map(12, 10);
30 path.clear();
31
32 // A snake-shaped path from the entry (left) to the exit (right).
33 for (int c = 0; c <= 10; ++c) path.push_back(std::make_pair(1, c));
34 for (int r = 2; r <= 4; ++r) path.push_back(std::make_pair(r, 10));
35 for (int c = 9; c >= 1; --c) path.push_back(std::make_pair(4, c));
36 for (int r = 5; r <= 7; ++r) path.push_back(std::make_pair(r, 1));
37 for (int c = 2; c <= 11; ++c) path.push_back(std::make_pair(7, c));
38
39 for (std::size_t i = 0; i < path.size(); ++i)
40 map.setCell(path[i].first, path[i].second, PATH);
41
42 map.setEntry(path.front().first, path.front().second);
43 map.setExit(path.back().first, path.back().second);
44}

92bool Game::placeTower(int row, int column, int type)
93{
94 if (state == GAME_OVER || state == VICTORY)
95 return false;
96
97 if (!map.isInside(row, column) || map.getCellType(row, column) != SCENERY)
98 {
99 message = "Towers can only be placed on green (scenery) cells.";
100 return false;

105 return false;

107

117 {
118 message = "Not enough coins for a " + placed.tower->getName() + ".";
119 return false;
120 }
121
122 placed.row = row;
123 placed.column = column;
124 placed.cooldown = 0.0;
125 placed.type = type;
126 message = "Placed a " + placed.tower->getName() + ".";
127 towers.push_back(std::move(placed));

201void Game::getCritterLocation(int index, double& row, double& column) const
202{
203 double position = wave.getCritter(index).getPosition();
204 if (position < 0.0) position = 0.0;
205 if (position > 100.0) position = 100.0;
206
207 double t = position / 100.0 * (static_cast<double>(path.size()) - 1.0);
208 std::size_t i = static_cast<std::size_t>(t);
209 if (i >= path.size() - 1)
210 {
211 row = path.back().first;
212 column = path.back().second;
213 return;
214 }
215 double f = t - static_cast<double>(i);
216 row = path[i].first + (path[i + 1].first - path[i].first) * f;
217 column = path[i].second + (path[i + 1].second - path[i].second) * f;
218}

220double Game::distanceToCritter(const PlacedTower& t, int critterIndex) const
221{
222 double r, c;
223 getCritterLocation(critterIndex, r, c);
224 double dr = r - t.row;
225 double dc = c - t.column;
226 return std::sqrt(dr * dr + dc * dc);

256void Game::fireTowers(double seconds)
257{
258 for (std::size_t k = 0; k < towers.size(); ++k)
259 {
260 PlacedTower& t = towers[k];
261 t.cooldown -= seconds;
262 if (t.cooldown > 0.0)
263 continue;
264
265 // Critters of the wave that are within range of this tower.
266 std::vector<int> inRange;
267 for (int i = 0; i < wave.getSize(); ++i)
268 if (isCritterOnMap(i) && distanceToCritter(t, i) <= t.tower->getRange())
269 inRange.push_back(i);
270
271 if (inRange.empty())
272 {
273 t.cooldown = 0.0; // ready as soon as something comes in range
274 continue;
275 }
276
277 // The critter closest to the exit is the main target.
278 int target = inRange[0];
279 for (std::size_t j = 1; j < inRange.size(); ++j)
280 if (wave.getCritter(inRange[j]).getPosition() >
281 wave.getCritter(target).getPosition())
282 target = inRange[j];
283
284 AreaDamageTower* area = dynamic_cast<AreaDamageTower*>(t.tower.get());
285 if (area != NULL)
286 {
287 std::vector<Critter*> targets;
288 for (std::size_t j = 0; j < inRange.size(); ++j)
289 {
290 targets.push_back(&wave.getCritter(inRange[j]));
291 double r, c;
292 getCritterLocation(inRange[j], r, c);
293 Shot s = { static_cast<double>(t.row), static_cast<double>(t.column), r, c, 0.12 };
294 shots.push_back(s);
295 }
296 area->attackMultiple(targets);
297 }
298 else
299 {
300 t.tower->attack(wave.getCritter(target));
301 double r, c;
302 getCritterLocation(target, r, c);
303 Shot s = { static_cast<double>(t.row), static_cast<double>(t.column), r, c, 0.12 };
304 shots.push_back(s);
305 }
306
307 t.cooldown = 1.0 / t.tower->getRateOfFire();
308 }
309}

331void Game::update(double seconds)
332{
333 for (std::size_t i = 0; i < shots.size();)
334 {
335 shots[i].timeLeft -= seconds;
336 if (shots[i].timeLeft <= 0.0) shots.erase(shots.begin() + i);
337 else ++i;
338 }
339
340 if (state != WAVE_RUNNING)
341 return;
342
343 // Release the next critter of the wave.
344 if (released < wave.getSize())
345 {
346 spawnTimer += seconds;
347 if (spawnTimer >= SPAWN_INTERVAL)
348 {
349 spawnTimer = 0.0;
350 ++released;
351 }
352 }
353
354 // Move the critters.
355 moveTimer += seconds;
356 while (moveTimer >= MOVE_INTERVAL && state == WAVE_RUNNING)
357 {
358 moveTimer -= MOVE_INTERVAL;
359 for (int i = 0; i < released; ++i)
360 {
361 if (!isCritterOnMap(i))
362 continue;
363 wave.getCritter(i).move();
364 if (wave.getCritter(i).getPosition() >= 100.0)
365 {
366 escapeCritter(i);
367 if (state != WAVE_RUNNING)
368 return;
369 }
370 }
371 }
372
373 fireTowers(seconds);
374 collectKills();
375 checkWaveEnd();
376}
