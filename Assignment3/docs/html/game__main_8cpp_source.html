92// 1) read what the player did (mouse clicks and key presses) and tell Game,
93// 2) ask Game to advance time (game.update) and draw its current state.

104 // What the player has chosen: the tower type to build with the next click,

120 // Keyboard: 1/2/3 choose a tower type, SPACE starts a wave,
121 // U levels up, P/B/F add an effect, T changes the targeting
122 // strategy, S sells the selected tower, R restarts, ESC quits.
123 else if (const auto* key = event->getIf<sf::Event::KeyPressed>())
124 {
125 switch (key->code)
126 {
127 case sf::Keyboard::Key::Num1: selectedType = Game::DIRECT_DAMAGE; break;
128 case sf::Keyboard::Key::Num2: selectedType = Game::AREA_DAMAGE; break;
129 case sf::Keyboard::Key::Num3: selectedType = Game::SLOW; break;
130 case sf::Keyboard::Key::Space: game.startWave(); break;
131 case sf::Keyboard::Key::U: game.upgradeTower(selectedRow, selectedColumn); break;
132 case sf::Keyboard::Key::P: game.addEffect(selectedRow, selectedColumn, Game::SPLASH); break;
133 case sf::Keyboard::Key::B: game.addEffect(selectedRow, selectedColumn, Game::BURNING); break;
134 case sf::Keyboard::Key::F: game.addEffect(selectedRow, selectedColumn, Game::FREEZING); break;
135 case sf::Keyboard::Key::T: game.cycleStrategy(selectedRow, selectedColumn); break;
136 case sf::Keyboard::Key::S:
137 if (game.sellTower(selectedRow, selectedColumn))
138 {
139 selectedRow = -1;
140 selectedColumn = -1;
141 }
142 break;
143 case sf::Keyboard::Key::R:
144 game.reset();
145 selectedRow = -1;
146 selectedColumn = -1;
147 break;
148 case sf::Keyboard::Key::Escape: window.close(); break;
149 default: break;
150 }
151 }
152 // Mouse: a left click on the map selects a tower or builds a new one.
153 else if (const auto* click = event->getIf<sf::Event::MouseButtonPressed>())
154 {

156 {
157 // Turn the pixel position of the click into a row and a column.
158 float fx = (click->position.x - MAP_X) / CELL;
159 float fy = (click->position.y - MAP_Y) / CELL;
160 int column = static_cast<int>(fx);
161 int row = static_cast<int>(fy);
162 bool inside = fx >= 0 && fy >= 0 &&
163 game.getMap().isInside(row, column);
164 if (inside)
165 {
166 // A tower is already there: just select it.
167 if (game.findTower(row, column) >= 0)
168 {
169 selectedRow = row;
170 selectedColumn = column;
171 }
172 // Empty cell: try to buy and place the chosen tower
173 // (Game checks that the cell is scenery and the player can pay).
174 else if (game.placeTower(row, column, selectedType))
175 {
176 selectedRow = row;
177 selectedColumn = column;
178 }
179 }
180 }
181 }
182 }
