42// Builds the one map of the game: a snake-shaped path with the entry on the
43// left and the exit on the right. The path cells are also saved, in walking
44// order, in the "path" list so a critter's position can be turned into a cell.
45void Game::buildMap()
46{
47 map = Map(12, 10);
48 path.clear();
49
50 // A snake-shaped path from the entry (left) to the exit (right).
51 for (int c = 0; c <= 10; ++c) path.push_back(std::make_pair(1, c));
52 for (int r = 2; r <= 4; ++r) path.push_back(std::make_pair(r, 10));
53 for (int c = 9; c >= 1; --c) path.push_back(std::make_pair(4, c));
54 for (int r = 5; r <= 7; ++r) path.push_back(std::make_pair(r, 1));
55 for (int c = 2; c <= 11; ++c) path.push_back(std::make_pair(7, c));
56
57 for (std::size_t i = 0; i < path.size(); ++i)
58 map.setCell(path[i].first, path[i].second, PATH);
59
60 map.setEntry(path.front().first, path.front().second);
61 map.setExit(path.back().first, path.back().second);
62}

114// Rules for placing a tower: the game must be running, the cell must be
115// scenery (Cell rule: towers only go on scenery), the cell must be empty,
116// and the player must be able to pay the buying cost.
117bool Game::placeTower(int row, int column, int type)
118{
119 if (state == GAME_OVER || state == VICTORY)
120 return false;
121
122 if (!map.isInside(row, column) || map.getCellType(row, column) != SCENERY)
123 {
124 message = "Towers can only be placed on green (scenery) cells.";
125 return false;
126 }
127 if (findTower(row, column) >= 0)
128 {
129 message = "There is already a tower on that cell.";
130 return false;
131 }
132
133 PlacedTower placed;
134 switch (type)
135 {
136 case AREA_DAMAGE: placed.tower.reset(new AreaDamageTower()); break;
137 case SLOW: placed.tower.reset(new SlowTower()); break;
138 default: placed.tower.reset(new DirectDamageTower()); type = DIRECT_DAMAGE; break;
139 }
140
141 if (!player.spendCoins(placed.tower->getBuyingCost()))
142 {
143 message = "Not enough coins for a " + placed.tower->getName() + ".";
144 return false;
145 }
146
147 placed.row = row;
148 placed.column = column;
149 placed.cooldown = 0.0;
150 placed.type = type;
151 message = "Placed a " + placed.tower->getName() + ".";
152 towers.push_back(std::move(placed));
153 return true;
154}

156// Decorator pattern: an upgrade does not change the tower, it wraps it in a
157// LevelUpDecorator (+1 level, +5 power, +1 range, higher refund). The price
158// (buying cost x current level) comes from Tower::getUpgradeCost(). The
159// maximum level is 3.
160bool Game::upgradeTower(int row, int column)
161{
162 if (state == GAME_OVER || state == VICTORY)
163 return false;
164
165 int index = findTower(row, column);
166 if (index < 0)
167 {
168 message = "Select a tower first.";
169 return false;
170 }
171 Tower& tower = *towers[index].tower;
172 if (tower.getLevel() >= tower.getMaxLevel())
173 {
174 message = "That tower is already at the maximum level.";
175 return false;
176 }
177 if (!player.spendCoins(tower.getUpgradeCost()))
178 {
179 message = "Not enough coins to upgrade.";
180 return false;
181 }
182 std::unique_ptr<Tower> upgraded(new LevelUpDecorator(std::move(towers[index].tower)));
183 towers[index].tower = std::move(upgraded);
184 message = "Upgraded to level " + std::to_string(towers[index].tower->getLevel()) + ".";
185 return true;
186}

188// Decorator pattern again: each effect wraps the tower in one more decorator.
189// A tower can have each effect only once.
190bool Game::addEffect(int row, int column, int effect)
191{
192 if (state == GAME_OVER || state == VICTORY)
193 return false;
194
195 int index = findTower(row, column);
196 if (index < 0)
197 {
198 message = "Select a tower first.";
199 return false;
200 }
201 Tower& tower = *towers[index].tower;
202 std::string name = effectName(effect);
203
204 if (tower.hasEffect(name))
205 {
206 message = "That tower already has " + name + ".";
207 return false;
208 }
209 if (effect == SPLASH && tower.attacksAll())
210 {
211 message = "Splash is useless on an Area Damage Tower: it already hits everything in range.";
212 return false;
213 }
214 if (!player.spendCoins(effectCost(effect)))
215 {
216 message = "Not enough coins for " + name + ".";
217 return false;
218 }
219
220 std::unique_ptr<Tower> decorated;
221 switch (effect)
222 {
223 case SPLASH: decorated.reset(new SplashDecorator(std::move(towers[index].tower))); break;
224 case BURNING: decorated.reset(new BurningDecorator(std::move(towers[index].tower))); break;
225 default: decorated.reset(new FreezingDecorator(std::move(towers[index].tower))); break;
226 }
227 towers[index].tower = std::move(decorated);
228 message = "Added " + name + ".";
229 return true;
230}

287// Asks the WaveFactory for the critters of the next wave (the more waves were
288// destroyed, the harder the wave) and remembers their
289// starting hit points (for the health bars). No critter is on the map yet:
290// update() releases them one by one.
291bool Game::startWave()
292{
293 if (state != PLANNING)
294 return false;
295
296 wave = waveFactory.createWave(waveNumber - 1);
297 status.assign(wave.getSize(), WALKING);
298 maxHitPoints.clear();
299 for (int i = 0; i < wave.getSize(); ++i)
300 maxHitPoints.push_back(wave.getCritter(i).getHitPoints());

303 spawnTimer = SPAWN_INTERVAL; // first critter appears immediately
304 moveTimer = 0.0;
305 state = WAVE_RUNNING;
306 message = "Wave " + std::to_string(waveNumber) + " started!";
307 return true;
308}

318// A critter stores its progress as a number from 0 (entry) to 100 (exit).
319// This turns that number into a point between two cells of the path,
320// so the window can draw it and the towers can measure the distance to it.
321void Game::getCritterLocation(int index, double& row, double& column) const
322{
323 double position = wave.getCritter(index).getPosition();
324 if (position < 0.0) position = 0.0;
325 if (position > 100.0) position = 100.0;
326
327 double t = position / 100.0 * (static_cast<double>(path.size()) - 1.0);
328 std::size_t i = static_cast<std::size_t>(t);
329 if (i >= path.size() - 1)
330 {
331 row = path.back().first;
332 column = path.back().second;
333 return;
334 }
335 double f = t - static_cast<double>(i);
336 row = path[i].first + (path[i + 1].first - path[i].first) * f;
337 column = path[i].second + (path[i + 1].second - path[i].second) * f;
338}

340// Straight-line distance, in cells, between two critters of the wave.
341double Game::distanceBetweenCritters(int first, int second) const
342{
343 double r1, c1, r2, c2;
344 getCritterLocation(first, r1, c1);
345 getCritterLocation(second, r2, c2);
346 return std::sqrt((r1 - r2) * (r1 - r2) + (c1 - c2) * (c1 - c2));
347}

349// Straight-line distance, in cells, between a tower and a critter.
350double Game::distanceToCritter(const PlacedTower& t, int critterIndex) const
351{
352 double r, c;
353 getCritterLocation(critterIndex, r, c);
354 double dr = r - t.row;
355 double dc = c - t.column;
356 return std::sqrt(dr * dr + dc * dc);
357}

389// Each tower has a cooldown (1 / rate of fire). When it is ready it collects
390// the critters inside its range and then:
391// - a tower that hits everything (attacksAll, the area damage tower) attacks
392// all of them,
393// - any other tower asks its targeting STRATEGY which critter to shoot, and
394// attacks it. The critters next to the target are passed on so that a
395// splash decorator can hit them too.
396// Burning and freezing are applied by the decorators inside the attack.
397void Game::fireTowers(double seconds)
398{
399 for (std::size_t k = 0; k < towers.size(); ++k)
400 {
401 PlacedTower& t = towers[k];
402 Tower& tower = *t.tower;
403 t.cooldown -= seconds;
404 if (t.cooldown > 0.0)
405 continue;
406
407 // Critters of the wave that are within range of this tower.
408 std::vector<TargetCandidate> candidates;
409 std::vector<int> candidateIndex;
410 for (int i = 0; i < wave.getSize(); ++i)
411 {
412 if (!isCritterOnMap(i))
413 continue;
414 double distance = distanceToCritter(t, i);
415 if (distance <= tower.getRange())
416 {
417 TargetCandidate candidate = { &wave.getCritter(i), distance };
418 candidates.push_back(candidate);
419 candidateIndex.push_back(i);
420 }
421 }
422
423 if (candidates.empty())
424 {
425 t.cooldown = 0.0; // ready as soon as something comes in range
426 continue;
427 }
428
429 if (tower.attacksAll())
430 {
431 std::vector<Critter*> targets;
432 for (std::size_t j = 0; j < candidates.size(); ++j)
433 {
434 targets.push_back(candidates[j].critter);
435 double r, c;
436 getCritterLocation(candidateIndex[j], r, c);
437 Shot s = { static_cast<double>(t.row), static_cast<double>(t.column), r, c, 0.12, false };
438 shots.push_back(s);
439 }
440 tower.attackAll(targets);
441 }
442 else
443 {
444 int chosen = tower.chooseTarget(candidates);
445 if (chosen < 0)
446 continue;
447 int targetIndex = candidateIndex[chosen];
448
449 // Critters near the target, for splash damage.
450 std::vector<Critter*> nearby;
451 std::vector<int> nearbyIndex;
452 double radius = tower.getSplashRadius();
453 if (radius > 0.0)
454 {
455 for (int j = 0; j < wave.getSize(); ++j)
456 if (j != targetIndex && isCritterOnMap(j) &&
457 distanceBetweenCritters(targetIndex, j) <= radius)
458 {
459 nearby.push_back(&wave.getCritter(j));
460 nearbyIndex.push_back(j);
461 }
462 }
463
464 tower.attackWithNearby(wave.getCritter(targetIndex), nearby);
465
466 double r, c;
467 getCritterLocation(targetIndex, r, c);
468 Shot s = { static_cast<double>(t.row), static_cast<double>(t.column), r, c, 0.12, false };
469 shots.push_back(s);
470
471 // One extra line from the target to each critter caught by the splash
472 // (it stays a little longer so the player can see it).
473 for (std::size_t j = 0; j < nearbyIndex.size(); ++j)
474 {
475 double nr, nc;
476 getCritterLocation(nearbyIndex[j], nr, nc);
477 Shot splashShot = { r, c, nr, nc, 0.3, true };
478 shots.push_back(splashShot);
479 }
480 }
481
482 t.cooldown = 1.0 / tower.getRateOfFire();
483 }
484}

486// A wave is over when every critter has been killed or has escaped.
487// Then the game is won (after the last wave) or goes back to PLANNING.
488void Game::checkWaveEnd()
489{
490 if (state != WAVE_RUNNING || released < wave.getSize())
491 return;
492 for (int i = 0; i < wave.getSize(); ++i)
493 if (status[i] == WALKING)
494 return;
495
496 if (waveNumber >= MAX_WAVES)
497 {
498 state = VICTORY;
499 message = "VICTORY! You survived all waves. Press R to play again.";
500 return;
501 }
502 ++waveNumber;
503 state = PLANNING;
504 message = "Wave cleared! Build or upgrade, then press SPACE for wave "
505 + std::to_string(waveNumber) + ".";
506}

508// Called every frame with the time since the last frame. Order of work:
509// 1) fade old shot lines, 2) release the next critter, 3) move the critters,
510// 4) towers shoot, 5) pay rewards for kills, 6) check whether the wave ended.
511void Game::update(double seconds)
512{
513 for (std::size_t i = 0; i < shots.size();)
514 {
515 shots[i].timeLeft -= seconds;
516 if (shots[i].timeLeft <= 0.0) shots.erase(shots.begin() + i);
517 else ++i;
518 }
519
520 if (state != WAVE_RUNNING)
521 return;
522
523 // Release the next critter of the wave.
524 if (released < wave.getSize())
525 {
526 spawnTimer += seconds;
527 if (spawnTimer >= SPAWN_INTERVAL)
528 {
529 spawnTimer = 0.0;
530 ++released;
531 }
532 }
533
534 // Burning and freezing run out with time (a burn can kill a critter).
535 for (int i = 0; i < released; ++i)
536 if (isCritterOnMap(i))
537 wave.getCritter(i).updateEffects(seconds);
538
539 // Move the critters.
540 moveTimer += seconds;
541 while (moveTimer >= MOVE_INTERVAL && state == WAVE_RUNNING)
542 {
543 moveTimer -= MOVE_INTERVAL;
544 for (int i = 0; i < released; ++i)
545 {
546 if (!isCritterOnMap(i))
547 continue;
548 wave.getCritter(i).move();
549 if (wave.getCritter(i).getPosition() >= 100.0)
550 {
551 escapeCritter(i);
552 if (state != WAVE_RUNNING)
553 return;
554 }
555 }
556 }
557
558 fireTowers(seconds);
559 collectKills();
560 checkWaveEnd();
561}
